class SearchNode:
        def __init__(self, state, parent=None, step_cost=0, cost=0, depth=0):
        self.state, self.parent = state, parent
        self.step_cost, self.cost, self.depth = step_cost, cost, depth
    def path(self):
        node, states = self, []
        while node:
            states.append(node.state)
            node = node.parent
        return states[::-1]
def successors(state):
    return ROMANIA_MAP.get(state, [])
def report(name, goal, expanded, max_fringe):
    print("\n" + "=" * 60)
    print(f"SOLUTION - {name}")
    print("=" * 60)
    if goal is None:
        print("No solution found!")

    print("Path Length :", goal.depth, "steps")
    print("Nodes Expanded:", expanded)
    print("Max Fringe :", max_fringe)
def bfs(start, goal):
    frontier = deque([SearchNode(start)])
    explored, seen, expanded, max_fringe = set(), {start}, 0, 1
    print("BREADTH-FIRST SEARCH (BFS)\n" + "-" * 60)
    while frontier:
        max_fringe = max(max_fringe, len(frontier))
        node = frontier.popleft()
        print(f"Expanding: {node.state:<15} (Depth: {node.depth})")
        if node.state == goal:
            return node, expanded, max_fringe
        explored.add(node.state)
        expanded += 1
        for nxt, cost in successors(node.state):
            if nxt not in explored and nxt not in seen:
                seen.add(nxt)
                frontier.append(SearchNode(nxt, node, cost, node.cost + cost, node.depth + 1))
    return None, expanded, max_fringe

6. Bayesian Network - Medical Diagnosis System
import itertools
import numpy as np
from typing import Dict, List, Tuple, Any, Optional
from collections import defaultdict
class BayesianNetwork:
        def __init__(self, name: str = "Bayesian Network"):
        self.name = name
        self.nodes = {}
        self.variables = []
        self.adjacency_list = {}
    def add_node(self, name: str, states: List[str], parents: List[str] = None,
                cpt: Dict[Tuple, float] = None):
                self.nodes[name] = {
            'states': states,
            'parents': parents if parents else [],
            'cpt': cpt if cpt else {},
            'children': []
        }
        if name not in self.adjacency_list:
            self.adjacency_list[name] = []
        if name not in self.variables:
            self.variables.append(name)
        if parents:
            for parent in parents:
                if parent not in self.adjacency_list:
                    self.adjacency_list[parent] = []
                if name not in self.adjacency_list[parent]:
                    self.adjacency_list[parent].append(name)
    def get_probability(self, node: str, state: str, parent_states: Dict[str, str]) -> float:
                node_info = self.nodes[node]
        parents = node_info['parents']
        if not parents:
            cpt = node_info['cpt']
            return cpt.get(state, 0.0)
        else:
            cpt = node_info['cpt']
            key = (state,) + tuple(parent_states[p] for p in parents)
            return cpt.get(key, 0.0)
    def get_all_states(self) -> Dict[str, List[str]]:
                return {node: self.nodes[node]['states'] for node in self.variables}
    def generate_all_assignments(self) -> List[Dict[str, str]]:
                all_states = self.get_all_states()
        variable_names = self.variables
        state_lists = [all_states[v] for v in variable_names]
        assignments = []
        for combination in itertools.product(*state_lists):
            assignment = dict(zip(variable_names, combination))
            assignments.append(assignment)
        return assignments
    def joint_probability(self, assignment: Dict[str, str]) -> float:
                prob = 1.0
        for node in self.variables:
            state = assignment[node]
            parents = self.nodes[node]['parents']
            parent_states = {p: assignment[p] for p in parents}
            prob *= self.get_probability(node, state, parent_states)
        return prob
    def marginal_probability(self, query_var: str, query_state: str,
                            evidence: Optional[Dict[str, str]] = None) -> float:
                evidence = evidence or {}
        all_assignments = self.generate_all_assignments()
        numerator = 0.0
        denominator = 0.0
        for assignment in all_assignments:
            matches_evidence = True
            for ev_var, ev_state in evidence.items():
                if assignment[ev_var] != ev_state:
                    matches_evidence = False
                    break
            if not matches_evidence:
                continue
            prob = self.joint_probability(assignment)
            if assignment[query_var] == query_state:
                numerator += prob
            denominator += prob
        if denominator == 0:
            return 0.0
        return numerator / denominator
    def query(self, query_var: str, evidence: Optional[Dict[str, str]] = None) -> Dict[str, float]:
                evidence = evidence or {}
        states = self.nodes[query_var]['states']
        result = {}
        for state in states:
            result[state] = self.marginal_probability(query_var, state, evidence)
        total = sum(result.values())
        if total > 0:
            for state in result:
                result[state] /= total
        return result
    def print_network_structure(self):
                print("\n" + "="*70)
        print(f" BAYESIAN NETWORK: {self.name}")
        print("="*70)
        print("\nVariables:")
        for var in self.variables:
            parents = self.nodes[var]['parents']
            children = self.adjacency_list.get(var, [])
            states = self.nodes[var]['states']
            print(f"\n[{var}]")
            print(f" States: {states}")
            print(f" Parents: {parents if parents else 'None (Prior)'}")
            print(f" Children: {children if children else 'None'}")
    def print_cpt(self, node: str):
                print(f"\nCPT for {node}:")
        print("-"*50)
        node_info = self.nodes[node]
        states = node_info['states']
        parents = node_info['parents']
        cpt = node_info['cpt']
        if not parents:
            print("Prior Probability:")
            for state in states:
                prob = cpt.get(state, 0.0)
                print(f" P({node} = {state}) = {prob:.3f}")
        else:
            parent_states = [self.nodes[p]['states'] for p in parents]
            print(f"Conditional on: {parents}")
            print(f"P({node} | {', '.join(parents)})")
            print("-"*50)
            for combo in itertools.product(*parent_states):
                parent_dict = dict(zip(parents, combo))
                parent_str = ', '.join([f"{p}={combo[i]}" for i, p in enumerate(parents)])
                print(f"\n Given: {parent_str}")
                    for state in states:

                    if pattern.name != fact.name:
                return None
            if len(pattern.args) != len(fact.args):
                return None
            substitution = {}
            for p_arg, f_arg in zip(pattern.args, fact.args):
                if p_arg.isupper() or p_arg.startswith('?'):
                    if p_arg in substitution:
                        if substitution[p_arg] != f_arg:
                            return None

                        substitution[p_arg] = f_arg
                elif p_arg != f_arg:
                    return None
            return substitution
    def apply_rule_forward(self, rule: Rule) -> List[Fact]:
                    conclusions = []
            premise_matches = []
            for premise in rule.premises:
                matches = []
                for fact in self.facts:
                    if self.fact_matches(fact, premise):
                        matches.append(fact)
                if not matches:
                    return []
                premise_matches.append(matches)
            for combination in itertools.product(*premise_matches):
                substitution = {}
                valid = True
                    for premise, fact in zip(rule.premises, combination):
                        sub = self.unify(premise, fact)
                        if sub is None:
                            valid = False
                            break
                        substitution.update(sub)
                    if valid:
                        conclusion = self.substitute(rule.conclusion, substitution)
                        if conclusion not in self.facts:
                            conclusions.append(conclusion)
            return conclusions
    def forward_chaining(self, max_iterations: int = 100) -> Set[Fact]:
                    derived = set()
            iteration = 0
            while iteration < max_iterations:
                new_facts = []
                for rule in self.rules:
                        new_conclusions = self.apply_rule_forward(rule)
                        for conclusion in new_conclusions:
                            if conclusion not in self.facts:
                                new_facts.append(conclusion)
                    if not new_facts:
                        break
                    for fact in new_facts:
                        self.facts.add(fact)
                        derived.add(fact)
                        self.inference_count += 1
                    iteration += 1
            return derived

                    if visited is None:
                visited = set()
            for fact in self.facts:
                    if self.fact_matches(fact, goal):
                        return True

        def explain(self, fact: Fact) -> str:
                        if fact in self.facts:
                    return f"'{fact}' is an initial fact (provided as input)."
            if fact not in self.working_memory:
                    return f"'{fact}' is not in working memory."
            for rule in self.rules:
                if rule.conclusion == fact:
                            return (f"'{fact}' was derived using rule: {rule}")
            for rule in self.rules:
                bindings = rule.conclusion.matches(fact)
                if bindings is not None:
                            premises_str = " AND ".join(
                                str(Fact(p.name, [bindings.get(arg, arg) for arg in p.args]))
                                for p in rule.premises
                            )
                            conclusion = Fact(fact.name,
                                            [bindings.get(arg, arg) for arg in rule.conclusion.args])
                            return (f"'{fact}' was derived using rule: "
                                f"IF {premises_str} THEN {conclusion}")
            return f"'{fact}' is in working memory but source could not be identified."

        def generate_traffic_data(self, days: int = 30) -> pd.DataFrame:
                        data = []
            start_date = datetime.now() - timedelta(days=days)
            for location in self.locations:
                area_type = self.area_types[location]
                    base_patterns = {
                        "commercial": {"peak_hours": [9, 10, 17, 18], "base_volume": 150},
                        "highway": {"peak_hours": [8, 9, 17, 18], "base_volume": 300},
                        "industrial": {"peak_hours": [7, 8, 16, 17], "base_volume": 200},
                            "residential": {"peak_hours": [7, 8, 17, 18], "base_volume": 80},
                            "educational": {"peak_hours": [8, 9, 12, 13, 17], "base_volume": 120},
                            "medical": {"peak_hours": [9, 10, 14, 15], "base_volume": 100},
                            "entertainment": {"peak_hours": [12, 13, 19, 20], "base_volume": 130},
                            "transportation": {"peak_hours": [7, 8, 17, 18, 21, 22], "base_volume": 250}
                    }
                    pattern = base_patterns.get(area_type, {"peak_hours": [9, 17], "base_volume": 100})
                    for day in range(days):
                        current_date = start_date + timedelta(days=day)
                            day_of_week = current_date.weekday()
                            is_weekend = day_of_week >= 5
                            for hour in range(24):
                                base_volume = pattern["base_volume"]
                                daily_variation = np.random.uniform(0.8, 1.2)
                                hour_effect = 0.5
                                for peak_hour in pattern["peak_hours"]:
                                    hour_effect += 0.4 * np.exp(-((hour - peak_hour) ** 2) / 8)
                                weekend_factor = 0.7 if is_weekend else 1.0
                                weather_factor = np.random.uniform(0.9, 1.1)
                                event_factor = np.random.uniform(1.2, 1.8) if np.random.random() < 0.05 else 1.0
                                volume = base_volume * daily_variation * hour_effect * weekend_factor * weather_factor * event_factor
                                volume = max(0, int(volume))
                                max_speed = 80 if area_type == "highway" else 50
                                speed_factor = 1 - (volume / (base_volume * 3))
                                speed = max(5, max_speed * max(0.3, speed_factor) + np.random.uniform(-5, 5))
                                if volume > base_volume * 1.8:
                                    congestion = "heavy"
                                elif volume > base_volume * 1.2:
                                    congestion = "moderate"
                                else:
                                    congestion = "light"
                                data.append({
                                    'timestamp': current_date.replace(hour=hour, minute=0, second=0),
                                    'location': location,
                                    'area_type': area_type,
                                    'hour': hour,
                                    'day_of_week': day_of_week,
                                    'is_weekend': int(is_weekend),
                                    'volume': volume,
                                    'speed': round(speed, 1),
                                    'congestion_level': congestion,
                                    'day': day
                                })
            df = pd.DataFrame(data)
            return df
class TrafficPredictor:
            def __init__(self):
            self.models = {}
            self.scaler = StandardScaler()
            self.results = {}

        def prepare_features(self, df: pd.DataFrame) -> Tuple[pd.DataFrame, pd.Series]:
            le_location = LabelEncoder()
            le_area = LabelEncoder()
            le_congestion = LabelEncoder()
            df['location_encoded'] = le_location.fit_transform(df['location'])
            df['area_encoded'] = le_area.fit_transform(df['area_type'])
            df['congestion_encoded'] = le_congestion.fit_transform(df['congestion_level'])
            df['hour_sin'] = np.sin(2 * np.pi * df['hour'] / 24)
            df['hour_cos'] = np.cos(2 * np.pi * df['hour'] / 24)
            df['day_sin'] = np.sin(2 * np.pi * df['day_of_week'] / 7)
            df['day_cos'] = np.cos(2 * np.pi * df['day_of_week'] / 7)
            features = ['hour', 'day_of_week', 'is_weekend', 'location_encoded',
                                'area_encoded', 'hour_sin', 'hour_cos', 'day_sin', 'day_cos']
            X = df[features]
            y = df['volume']
            return X, y
        def train_random_forest(self, X_train: pd.DataFrame, y_train: pd.Series):
                        print("\nTraining Random Forest Regressor...")
            model = RandomForestRegressor(

            )
            model.fit(X_train, y_train)

        def train_gradient_boosting(self, X_train: pd.DataFrame, y_train: pd.Series):
                        print("\nTraining Gradient Boosting Regressor...")
            model = GradientBoostingRegressor(

        def train_neural_network(self, X_train: pd.DataFrame, y_train: pd.Series,
                                epochs: int = 50, batch_size: int = 64):
                        print("\nTraining Neural Network...")
            X_train_scaled = self.scaler.fit_transform(X_train.values)
            X_tensor = torch.FloatTensor(X_train_scaled)
            y_tensor = torch.FloatTensor(y_train.values).reshape(-1, 1)
            dataset = TensorDataset(X_tensor, y_tensor)
            dataloader = DataLoader(dataset, batch_size=batch_size, shuffle=True)
            class TrafficNN(nn.Module):
                def __init__(self, input_size):
                    super().__init__()
                    self.network = nn.Sequential(
                        nn.Linear(input_size, 128),
                                nn.ReLU(),
                                nn.Dropout(0.2),
                                nn.Linear(128, 64),
                                nn.ReLU(),
                                nn.Dropout(0.2),
                                nn.Linear(64, 32),
                                nn.ReLU(),
                                nn.Linear(32, 1)
                            )
                    def forward(self, x):
                        return self.network(x)
            model = TrafficNN(X_train.shape[1])
            criterion = nn.MSELoss()
            optimizer = optim.Adam(model.parameters(), lr=0.001)
            model.train()
            losses = []
            for epoch in range(epochs):
                epoch_loss = 0
                for X_batch, y_batch in dataloader:
                            optimizer.zero_grad()
                            predictions = model(X_batch)
                            loss = criterion(predictions, y_batch)
                            loss.backward()
                            optimizer.step()
                            epoch_loss += loss.item()
                    losses.append(epoch_loss / len(dataloader))
                    if (epoch + 1) % 10 == 0:
                        print(f" Epoch {epoch+1}/{epochs}, Loss: {losses[-1]:.4f}")
            self.models['neural_network'] = model
            return model, losses

                        print("\n" + "="*60)

                    with torch.no_grad():

                else:
                    predictions = model.predict(X_test)

                    mse = mean_squared_error(y_test, predictions)
                    rmse = np.sqrt(mse)
                    r2 = r2_score(y_test, predictions)
                    metrics[name] = {

                    }
                    print(f"\n{name.upper()}:")
                    print(f" MAE: {mae:.2f}")
                    print(f" RMSE: {rmse:.2f}")
                    print(f" R² Score: {r2:.3f}")
            self.performance_metrics = metrics
            return metrics

class HealthcareAIApplication:
            def __init__(self):
            self.data = None
            self.models = {}
            self.scaler = StandardScaler()
        def generate_healthcare_data(self, n_patients: int = 1000):
                        np.random.seed(42)
            data = {
                'patient_id': range(1, n_patients + 1),
                'age': np.random.randint(18, 90, n_patients),
                'gender': np.random.choice(['M', 'F'], n_patients),
                'bmi': np.random.normal(28, 6, n_patients),
                'blood_pressure': np.random.randint(100, 180, n_patients),
                'cholesterol': np.random.randint(150, 300, n_patients),
                'diabetes': np.random.choice([0, 1], n_patients, p=[0.8, 0.2]),
                'heart_disease': np.random.choice([0, 1], n_patients, p=[0.85, 0.15]),
                    'previous_admissions': np.random.poisson(0.5, n_patients),
                    'length_of_stay': np.random.exponential(5, n_patients),
                    'medications': np.random.randint(0, 6, n_patients),
                    'follow_up_appointments': np.random.choice([0, 1, 2], n_patients, p=[0.3, 0.4, 0.3])
            }
            df = pd.DataFrame(data)
            logit = (0.01 * df['age'] + 0.02 * df['bmi'] + 0.005 * df['blood_pressure'] +
                    0.1 * df['diabetes'] + 0.15 * df['heart_disease'] + 0.2 * df['previous_admissions'] +
                            0.05 * df['length_of_stay'] + 0.1 * df['medications'] - 0.05 * df['follow_up_appointments'] - 2)
            prob = 1 / (1 + np.exp(-logit))
            df['readmitted'] = np.random.binomial(1, prob)
            self.data = df
            return df
        def run_healthcare_demo(self):
                        print("\n" + "="*70)
            print(" HEALTHCARE AI APPLICATION - PATIENT READMISSION PREDICTOR")
            print("="*70)
            df = self.generate_healthcare_data(1000)
            print(f"\n✓ Generated {len(df)} patient records")
            print(f"✓ Readmission Rate: {df['readmitted'].mean()*100:.1f}%")
            features = ['age', 'bmi', 'blood_pressure', 'cholesterol', 'diabetes',
                        'heart_disease', 'previous_admissions', 'length_of_stay',
                        'medications', 'follow_up_appointments']
            X = df[features]
            y = df['readmitted']
            X_scaled = self.scaler.fit_transform(X)
            X_train, X_test, y_train, y_test = train_test_split(
                X_scaled, y, test_size=0.2, random_state=42
            )
            from sklearn.ensemble import RandomForestClassifier
            from sklearn.metrics import classification_report, roc_auc_score
            model = RandomForestClassifier(n_estimators=100, random_state=42, n_jobs=-1)
            model.fit(X_train, y_train)
            self.models['random_forest'] = model
            y_pred = model.predict(X_test)
            y_prob = model.predict_proba(X_test)[:, 1]
            print("\n📊 MODEL EVALUATION:")
            print("-"*40)
            print(classification_report(y_test, y_pred))
            print(f"AUC-ROC: {roc_auc_score(y_test, y_prob):.3f}")
            feature_importance = pd.DataFrame({
                'feature': features,
                'importance': model.feature_importances_
            }).sort_values('importance', ascending=False)
            print("\n📈 TOP PREDICTIVE FACTORS:")
            for _, row in feature_importance.head(5).iterrows():
                    print(f" • {row['feature']}: {row['importance']:.3f}")
            return model
